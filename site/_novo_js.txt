/* ---------- mapa esquemático (SVG, sem tiles) ---------- */
const NS = "http://www.w3.org/2000/svg";
const sv = (tag, attrs = {}, ...kids) => { const n = document.createElementNS(NS, tag); for (const [k, v] of Object.entries(attrs)) if (v != null && v !== false) n.setAttribute(k, v); for (const c of kids.flat()) if (c) n.append(c); return n; };
const MCOL = { trilha: "var(--c-academia)", praia: "var(--c-dia)", mirante: "var(--c-cedo)", noite: "var(--c-noite)", comida: "var(--c-sunset)", atividade: "var(--c-prova)", passeio: "var(--c-estrada)", evento: "var(--c-prova)", "bate-volta": "var(--c-estrada)" };
const comCoord = l => l.coords && l.coords.lat != null;
const hav = (a, b) => { const R = 6371000, r = Math.PI / 180, dp = (b[0] - a[0]) * r, dl = (b[1] - a[1]) * r; const h = Math.sin(dp / 2) ** 2 + Math.cos(a[0] * r) * Math.cos(b[0] * r) * Math.sin(dl / 2) ** 2; return 2 * R * Math.asin(Math.sqrt(h)); };
const dtxt = m => m < 950 ? `${Math.round(m / 50) * 50 || 50} m` : `${(m / 1000).toFixed(m < 10000 ? 1 : 0).replace(".", ",")} km`;
function bboxDe(pts, pad = 0.1, minKm = 0) {
  let w = 1e9, e = -1e9, s = 1e9, n = -1e9;
  for (const [lng, lat] of pts) { w = Math.min(w, lng); e = Math.max(e, lng); s = Math.min(s, lat); n = Math.max(n, lat); }
  const cl = (n + s) / 2, cg = (e + w) / 2, minLat = minKm / 111.32, minLng = minKm / (111.32 * Math.cos(cl * Math.PI / 180));
  const hl = Math.max((n - s) / 2 * (1 + pad), minLat / 2), hg = Math.max((e - w) / 2 * (1 + pad), minLng / 2);
  return { w: cg - hg, e: cg + hg, s: cl - hl, n: cl + hl };
}
const ILHA_BB = bboxDe(D.ilha, 0.03);
function proj(bb, W, H) {
  const c = Math.cos((bb.s + bb.n) / 2 * Math.PI / 180), dx = (bb.e - bb.w) * c, dy = bb.n - bb.s, k = Math.min(W / dx, H / dy), ox = (W - dx * k) / 2, oy = (H - dy * k) / 2;
  return { x: lng => ox + (lng - bb.w) * c * k, y: lat => oy + (bb.n - lat) * k, k };
}
const caminho = (pts, P, fechar) => "M" + pts.map(p => `${P.x(p[0]).toFixed(1)},${P.y(p[1]).toFixed(1)}`).join("L") + (fechar ? "Z" : "");
function mapaSvg({ bb, W, H, marcas, trilhas, destaque, selecionado, rotulos, onSel, rotuloAria }) {
  const P = proj(bb, W, H), s = sv("svg", { viewBox: `0 0 ${W} ${H}`, role: "img", "aria-label": rotuloAria || "Mapa esquemático", class: "map" });
  s.append(sv("rect", { width: W, height: H, class: "m-mar" }), sv("path", { d: caminho(D.ilha, P, true), class: "m-terra" }));
  if (trilhas) for (const t of trilhas) for (const ln of t.linhas) s.append(sv("path", { d: caminho(ln, P), class: "m-trilha" + (t._d ? " m-trilha-d" : "") }));
  const lab = [];
  for (const l of marcas) {
    const x = P.x(l.coords.lng), y = P.y(l.coords.lat); if (x < -10 || y < -10 || x > W + 10 || y > H + 10) continue;
    const d = l.id === destaque, sel = l.id === selecionado;
    const g = sv("g", { class: "m-pt" + (d || sel ? " on" : ""), tabindex: "0", role: "link", "aria-label": l.nome, style: "cursor:pointer" });
    g.append(sv("circle", { cx: x, cy: y, r: 15, fill: "transparent" }), sv("circle", { cx: x, cy: y, r: d || sel ? 8 : 5.5, fill: MCOL[l.categoria] || "var(--sea)", class: "m-c" }));
    const go = () => { if (onSel) onSel(l.id); else location.hash = "lugar-" + l.id; };
    g.addEventListener("click", go); g.addEventListener("keydown", e => { if (e.key === "Enter" || e.key === " ") { e.preventDefault(); go(); } });
    g.append(sv("title", {}, document.createTextNode(l.nome)));
    s.append(g);
    if (rotulos || d || sel) lab.push([x, y, l.nome.replace(/\s*\(.*\)$/, ""), d || sel]);
  }
  for (const [x, y, t, forte] of lab) { const right = x < W * 0.62; const tx = sv("text", { x: x + (right ? 11 : -11), y: y + 4, "text-anchor": right ? "start" : "end", class: "m-lab" + (forte ? " forte" : "") }); tx.textContent = t.length > 26 ? t.slice(0, 25) + "…" : t; s.append(tx); }
  return s;
}
function trilhasDe(id) { return D.trilhas.filter(t => t.lugar === id); }
